## 4. API Route：/api/speech/feedback —— 编排主链路

```ts
// app/api/speech/feedback/route.ts
import { NextRequest, NextResponse } from "next/server";
import { generateObject } from "ai";
import { getChatModel, FIXED_PROVIDER_OPTIONS, FIXED_GEN_CONFIG } from "@/lib/ai/provider";
import { SpeakingFeedbackSchema } from "@/lib/ai/schemas/speaking";
import { ACTIVE_SPEAKING_EVAL } from "@/lib/ai/prompts/speaking-eval";
import { checkDailyQuota } from "@/lib/ai/usage";
import { withRetryOnce } from "@/lib/ai/retry";
import { scanOutput } from "@/lib/ai/guard";
import { logAiCall } from "@/lib/ai/logger";
import { createClient } from "@/lib/supabase/server";

export const maxDuration = 60; // 本地 node 进程无函数时长限制；保留 60s 兜底降级（兼容潜在云迁移）

// v2.4 单用户：无认证，直接服务 seed 预设的唯一学生
const STUDENT_ID = process.env.SEED_STUDENT_ID!;

export async function POST(req: NextRequest) {
  const { lessonId, submissionId, transcript, durationMs, pauses } = await req.json();
  const supabase = await createClient();
  const user = { id: STUDENT_ID };

  // 配额：只拦学生主动调用；系统重试与测试账号豁免（修复 S-5）
  const quota = await checkDailyQuota(user.id, { source: "student" });
  if (!quota.ok) return NextResponse.json({ status: "quota_exceeded", resetsAt: quota.resetsAt });

  const lesson = await loadLessonTaskCard(supabase, lessonId); // 任务卡+句型框架
  const prompt = ACTIVE_SPEAKING_EVAL.build({
    taskCard: lesson.skill_goal, sentenceFrames: lesson.sentence_frames,
    transcript, durationMs, pauseCount: pauses,
  });

  const startedAt = Date.now();
  const result = await withRetryOnce(() =>
    generateObject({
      model: getChatModel(),
      schema: SpeakingFeedbackSchema,
      messages: prompt,
      providerOptions: FIXED_PROVIDER_OPTIONS,
      ...FIXED_GEN_CONFIG,
      temperature: 0, // 重试路径由 retry.ts 保证 t=0
    })
  );

  if (!result.ok) {
    await markPending(supabase, submissionId);          // pending，不阻塞过关
    await logAiCall({ userId: user.id, kind: "speaking_eval",
      promptVersion: ACTIVE_SPEAKING_EVAL.version, status: "failed",
      latencyMs: Date.now() - startedAt, submissionId });
    return NextResponse.json({ status: "pending", message: "反馈稍后送达" });
  }

  const feedback = scanOutput(result.value);            // 输出侧护栏
  await persistFeedback(supabase, submissionId, feedback);
  await logAiCall({ userId: user.id, kind: "speaking_eval",
    promptVersion: ACTIVE_SPEAKING_EVAL.version, status: "ok",
    tokensIn: result.usage?.inputTokens, tokensOut: result.usage?.outputTokens,
    latencyMs: Date.now() - startedAt, submissionId }); // 不存原文，只存引用
  return NextResponse.json({ status: "ok", feedback });
}
```

## 5. components/useRecorder.ts —— 媒体采集（修复 I-3）

```ts
// components/useRecorder.ts —— React hook 骨架
"use client";
import { useRef, useState } from "react";

/** iOS Safari 只支持 mp4 系，Chrome 优先 webm/opus；硬编码 webm 会在 iOS 直接失败 */
export function pickMimeType(kind: "audio" | "video"): string {
  const candidates = kind === "audio"
    ? ["audio/mp4", "audio/webm;codecs=opus", "audio/webm"]
    : ["video/mp4", "video/webm;codecs=vp9,opus", "video/webm"];
  return candidates.find((t) => MediaRecorder.isTypeSupported(t)) ?? "";
}

export function useRecorder(kind: "audio" | "video", maxSec: number) {
  const mediaRef = useRef<MediaRecorder | null>(null);
  const chunksRef = useRef<Blob[]>([]);
  const [state, setState] = useState<"idle" | "recording" | "done" | "error">("idle");

  async function start() {
    const stream = await navigator.mediaDevices.getUserMedia(
      kind === "video"
        ? { video: { width: 960, height: 540, facingMode: "user" }, audio: true }
        : { audio: true },
    ).catch(() => null);
    if (!stream) { setState("error"); return; }          // 无摄像头/拒权限 → 上层降级 UI

    const mimeType = pickMimeType(kind);
    const rec = new MediaRecorder(stream, {
      mimeType,
      videoBitsPerSecond: kind === "video" ? 700_000 : undefined, // 修复 S-2：700Kbps
      audioBitsPerSecond: 64_000,
    });
    chunksRef.current = [];
    rec.ondataavailable = (e) => e.data.size && chunksRef.current.push(e.data);
    rec.onstop = () => setState("done");
    rec.start(1_000);
    mediaRef.current = rec;
    setState("recording");
    setTimeout(() => rec.state === "recording" && rec.stop(), maxSec * 1_000); // 限时硬停
  }

  function stop(): Promise<{ blob: Blob; mimeType: string }> {
    return new Promise((resolve) => {
      const rec = mediaRef.current!;
      rec.onstop = () => {
        const mimeType = rec.mimeType;
        resolve({ blob: new Blob(chunksRef.current, { type: mimeType }), mimeType });
        setState("done");
        rec.stream.getTracks().forEach((t) => t.stop());
      };
      rec.stop();
    });
  }
  return { state, start, stop };
}
```

## 6. 视频学习器：YouTube 切片 + 字幕同步骨架

```tsx
// components/VideoLearner.tsx 骨架（依赖 react-youtube）
"use client";
import YouTube, { YouTubePlayer } from "react-youtube";
import { useEffect, useRef, useState } from "react";

type Subtitle = { start: number; end: number; text: string };

export function VideoLearner({ video, subtitles, onEmbedError }: {
  video: { youtube_id: string; start_sec: number; end_sec: number };
  subtitles: Subtitle[];
  onEmbedError: (code: number) => void; // 101/150 = 禁止嵌入 → 上报 videos.status（修复 I-4）
}) {
  const playerRef = useRef<YouTubePlayer | null>(null);
  const [activeIdx, setActiveIdx] = useState(-1);

  useEffect(() => {
    const timer = setInterval(async () => {
      const t = await playerRef.current?.getCurrentTime();
      if (t == null) return;
      setActiveIdx(subtitles.findIndex((s) => t >= s.start && t < s.end));
      if (t >= video.end_sec) playerRef.current?.pauseVideo(); // 切片右边界
    }, 250);
    return () => clearInterval(timer);
  }, [subtitles, video.end_sec]);

  return (
    <div>
      <YouTube
        videoId={video.youtube_id}
        opts={{ playerVars: { start: video.start_sec, end: video.end_sec, rel: 0 } }}
        onReady={(e) => { playerRef.current = e.target; }}
        onError={(e) => onEmbedError(e.data)}
      />
      <div className="subtitle-track">
        {subtitles.map((s, i) => (
          <p key={i} data-active={i === activeIdx}
             onClick={() => playerRef.current?.seekTo(s.start, true)}>
            {s.text /* 点查：按空格切词，命中术语词表弹释义卡 */}
          </p>
        ))}
      </div>
    </div>
  );
}
```

## 7. lib/review/scheduler.ts —— SM-2 简化版

```ts
// lib/review/scheduler.ts
type Familiarity = 0 | 1 | 2; // 陌生 / 模糊 / 掌握（FR-4.4）

/** 三档 → 下次复习间隔（天）。比完整 SM-2 简化：无 EF 因子，固定阶梯。 */
export function nextReviewAt(f: Familiarity, streak: number, now = new Date()): Date {
  const intervals: Record<Familiarity, number[]> = {
    0: [0.02, 1, 1, 2],        // 陌生：30分钟后 → 1天 → 1天 → 2天（重置连对）
    1: [1, 2, 4, 7],           // 模糊
    2: [2, 4, 7, 15, 30],      // 掌握
  };
  const ladder = intervals[f];
  const days = ladder[Math.min(streak, ladder.length - 1)];
  const next = new Date(now);
  next.setDate(next.getDate() + Math.floor(days));
  next.setHours(next.getHours() + Math.round((days % 1) * 24));
  return next;
}

export function nextStreak(f: Familiarity, streak: number): number {
  return f === 0 ? 0 : streak + 1; // 陌生清零连对，其余累进
}
```

## 8. scripts/import-content.ts —— 内容灌装（修复 I-5）

```ts
// scripts/import-content.ts —— CSV → Zod 校验 → 幂等 upsert
import { parse } from "csv-parse/sync";
import { z } from "zod";
import { createClient } from "@supabase/supabase-js";
import * as fs from "node:fs";

const ChunkRow = z.object({
  unit_no: z.coerce.number().int().min(1).max(8),
  item_no: z.coerce.number().int(),            // 幂等键组成部分
  text: z.string().min(2),
  meaning: z.string().min(1),
  example: z.string().min(5),
});
type ChunkRow = z.infer<typeof ChunkRow>;

async function main(file: string) {
  const rows: unknown[] = parse(fs.readFileSync(file, "utf8"), { columns: true });
  const valid: ChunkRow[] = [];
  const errors: string[] = [];
  rows.forEach((r, i) => {
    const p = ChunkRow.safeParse(r);
    p.success ? valid.push(p.data) : errors.push(`行 ${i + 2}: ${p.error.message}`);
  });
  if (errors.length) { console.error(errors.join("\n")); process.exit(1); } // 全过才导入

  const db = createClient(process.env.SUPABASE_URL!, process.env.SUPABASE_SERVICE_ROLE_KEY!);
  const { error } = await db.from("chunks").upsert(valid, {
    onConflict: "unit_no,item_no",             // 唯一约束 (unit_no, item_no)
  });
  if (error) throw error;
  console.log(`upserted ${valid.length} rows (idempotent)`);
}
main(process.argv[2]!);
```

## 10. lib/time.ts + lib/ai/usage.ts —— 北京时间日界与配额口径（NFR-8 / 修复 S-5）

```ts
// lib/time.ts —— NFR-8：所有"每日"统计（QUOTA-1 配额 / QUOTA-5 限流 / streaks 打卡 / 复习队列）的日界一律北京时间
/**
 * 固定 +8 偏移计算北京时间当日 0 点（返回 UTC 时刻）。
 * 关键：禁止 new Date().setHours(0,0,0,0)——那是服务器本地时区（Linux 服务器时区可能非 UTC+8），
 * 学生北京时间晚间学习会被算进"明天"，导致配额/打卡跨日错算（NFR-8 红线）。
 */
export function beijingDayStart(now = new Date()): Date {
  const beijing = new Date(now.getTime() + 8 * 3_600_000);   // 平移到北京时间轴
  beijing.setUTCHours(0, 0, 0, 0);                           // 北京时间当日 0 点
  return new Date(beijing.getTime() - 8 * 3_600_000);        // 换回 UTC 时刻
}

export function nextBeijingDayStart(now = new Date()): Date {
  return new Date(beijingDayStart(now).getTime() + 86_400_000);
}
```

```ts
// lib/ai/usage.ts
import { createClient } from "@/lib/supabase/server";
import { beijingDayStart, nextBeijingDayStart } from "@/lib/time";

const DAILY_STUDENT_LIMIT = 3;

/** 只统计学生主动触发的成功/失败调用；系统重试（source='system'）不计 */
export async function checkDailyQuota(userId: string, { source }: { source: "student" | "system" }) {
  if (source === "system") return { ok: true as const };
  const supabase = await createClient();
  const { data: profile } = await supabase.from("users").select("role").eq("id", userId).single();
  if (profile?.role === "admin" || profile?.role === "test") return { ok: true as const }; // 验收豁免

  const since = beijingDayStart();                          // NFR-8：勿用 setHours（本地时区）
  const { count } = await supabase.from("ai_call_logs")
    .select("*", { count: "exact", head: true })
    .eq("user_id", userId).eq("source", "student")
    .gte("created_at", since.toISOString());
  const used = count ?? 0;
  return used < DAILY_STUDENT_LIMIT
    ? { ok: true as const }
    : { ok: false as const, resetsAt: nextBeijingDayStart() };
}
```

## 11. API Route：/api/quiz/submit —— 问答判分（v2.2 新增，API-15，全程不调 LLM）

```ts
// app/api/quiz/submit/route.ts
import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@/lib/supabase/server";

type Answer = { questionId: string; answer: string };

export async function POST(req: NextRequest) {
  const { lessonId, phase, answers } = await req.json() as {
    lessonId: string; phase: "L1_predict" | "L2_comprehension"; answers: Answer[];
  };
  const supabase = await createClient();
  const userId = process.env.SEED_STUDENT_ID!; // v2.4 单用户：无认证，直接服务 seed 学生

  const { data: questions } = await supabase
    .from("quiz_questions")
    .select("id, answer, explanation")
    .eq("lesson_id", lessonId).eq("phase", phase);

  if (phase === "L1_predict") {
    // 开放题：只落库不判分；同一事务把作答复制进 problem_logs 作 L5 写作素材（FR-4.6）
    const { error } = await supabase.rpc("submit_l1_quiz", { p_user: userId, p_lesson: lessonId, p_answers: answers });
    if (error) return NextResponse.json({ error: "db_error" }, { status: 500 });
    return NextResponse.json({ results: null });            // 恒不判分
  }

  // L2_comprehension：服务端对固定答案库即时判分（无 LLM、无重试链路）
  const results = (questions ?? []).map((q) => ({
    questionId: q.id,
    isCorrect: q.answer === answers.find((a) => a.questionId === q.id)?.answer,
    explanation: q.explanation,
  }));
  await supabase.from("quiz_submissions").insert(answers.map((a) => ({
    user_id: userId, question_id: a.questionId, answer_text: a.answer,
    is_correct: results.find((r) => r.questionId === a.questionId)?.isCorrect ?? null,
  })));
  return NextResponse.json({ results });                    // 答错不阻塞过关（LC-L2）
}
```

> L1 的"落库 + 素材复制"两步用 Postgres 函数 `submit_l1_quiz` 包成单事务（原子性断言见《测试方案.md》§3.2）；rpc 体内就是两条 insert，写在 0003 迁移里。

## 12. 前端错误上报 —— 全局捕获 + 静默落库（v2.2 新增，NFR-5 / API-19 / DB-26）

```tsx
// components/ErrorReporter.tsx —— 根布局挂载一次
"use client";
import { useEffect } from "react";

export function ErrorReporter({ userId }: { userId?: string }) {
  useEffect(() => {
    const report = (message: string, stack?: string) => {
      fetch("/api/errors/report", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          message: message.slice(0, 500),
          stack: stack?.slice(0, 2048),          // DB-26：截断 ≤2KB
          url: location.href,
          userAgent: navigator.userAgent,
          userId,                                // 可空：未登录也允许上报
        }),
        keepalive: true,                         // 页面卸载途中也尽力送达
      }).catch(() => {});                        // 静默失败，绝不打扰用户（NFR-5）
    };
    const onError = (e: ErrorEvent) => report(e.message, e.error?.stack);
    const onRejection = (e: PromiseRejectionEvent) =>
      report(`unhandledrejection: ${String(e.reason)}`);
    window.addEventListener("error", onError);
    window.addEventListener("unhandledrejection", onRejection);
    return () => {
      window.removeEventListener("error", onError);
      window.removeEventListener("unhandledrejection", onRejection);
    };
  }, [userId]);
  return null;
}
// React 组件树崩溃由 error boundary 捕获后调用同一 report 函数（复用截断与静默逻辑）
```

```ts
// app/api/errors/report/route.ts
import { NextRequest, NextResponse } from "next/server";
import { createClient } from "@/lib/supabase/server";

export async function POST(req: NextRequest) {
  try {
    const { message, stack, url, userAgent, userId } = await req.json();
    const supabase = await createClient();
    await supabase.from("error_logs").insert({
      user_id: userId ?? null,
      message: String(message).slice(0, 500),
      stack: stack ? String(stack).slice(0, 2048) : null,
      url, user_agent: userAgent,
      released_version: process.env.NEXT_PUBLIC_APP_VERSION ?? null,
    });
  } catch { /* 静默失败：排障通道自身永不制造用户可见错误 */ }
  return NextResponse.json({ status: "ok" });
}
```

```sql
-- L1 提交原子化：作答落库 + 素材复制进 problem_logs 同事务（API-15 / FR-4.6）
create or replace function submit_l1_quiz(p_user uuid, p_lesson uuid, p_answers jsonb)
returns void language plpgsql security definer as $$
begin
  insert into quiz_submissions (user_id, question_id, answer_text, is_correct)
  select p_user, (a->>'questionId')::uuid, a->>'answer', null
  from jsonb_array_elements(p_answers) a;

  insert into problem_logs (user_id, lesson_id, source, content)
  select p_user, p_lesson, 'quiz_l1', a->>'answer'
  from jsonb_array_elements(p_answers) a;
end $$;

-- RLS
alter table quiz_questions enable row level security;
alter table quiz_submissions enable row level security;
alter table notes enable row level security;
alter table content_versions enable row level security;
alter table error_logs enable row level security;

-- 题库全员只读（判分在服务端做，answer 不下发前端——由 API Route 按需取）
-- v2.4 单用户：固定 user_id 过滤（GUC app.seed_student_id，迁移时 set）
create policy "read questions" on quiz_questions
  for select using (true);
create policy "own submissions" on quiz_submissions
  for all using (user_id = current_setting('app.seed_student_id')::uuid)
  with check (user_id = current_setting('app.seed_student_id')::uuid);
```
